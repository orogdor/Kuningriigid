
  function fortificationMasonry(a,b,bottom,top,width,seed,bronze){
    mesoCoursesV361(a,b,bottom,top,width,seed,bronze,
      bottom<.15&&top>1.0?.12:0);
  }
  function highStoneWall(from,to,index,access,joins){
    const len=from.distanceTo(to);
    if(len<.08)return;
    const axis=to.clone().sub(from).normalize();
    const side=V(-axis.z,0,axis.x),heading=-Math.atan2(axis.z,axis.x);
    const {width,deck,parapet}=wallWalk,bronze=lastWallStyle!=='cutstone'&&lastWallStyle!=='limestone';
    // Battered masonry base (wider at ground), with real running stone
    // courses. A single flat colored core/pebble texture is never exposed.
    mesoCoursesV361(from,to,-.09,deck-.10,width,index+49,bronze,.43,true);
    const paving=new T.MeshStandardMaterial({color:0xcab48e,roughness:1});
    block(building,len+.015,.15,width+.06,paving,
      from.clone().lerp(to,.5).setY(deck-.075),heading,.93)
      .name='V361 fortress continuous stone wall walk';
    wallWalkBand(from,to,width*.85,deck,deck);
    const jointMat=new T.MeshStandardMaterial({color:0x9d8667,roughness:1});
    const joints=Math.max(1,Math.ceil(len/.92));
    for(let i=1;i<joints;i++){
      block(building,.018,.013,width*.88,jointMat,
        from.clone().lerp(to,i/joints).setY(deck+.009),heading)
        .name='V361 walkway stone coursing joint';
    }
    // Two thin plastered protective walls around a genuinely open central
    // walkway. Both crest with the rounded Babylonian waveform.
    for(const sign of [-1,1]){
      const edge=side.clone().multiplyScalar(sign*(width*.50+.085));
      const A=from.clone().add(edge),B=to.clone().add(edge);
      const begin=0,end=1; // joined modules overlap cleanly, no corner voids
      const segments=access?.side===sign
        ?[[begin,clamp((from.distanceTo(access.point)-access.gap)/len,begin,end)],
          [clamp((from.distanceTo(access.point)+access.gap)/len,begin,end),end]]
        :[[begin,end]];
      for(const [l,r] of segments){
        if(r-l<.001)continue;
        const a=A.clone().lerp(B,l),b=A.clone().lerp(B,r);
        mesoWaveV361(a,b,deck,.32,sign>0?1.01:.91,
          index*13+(sign>0?71:83),sign>0);
        walkBarrier(a,b,.15,deck-.04,deck+parapet+.11);
      }
    }
  }
